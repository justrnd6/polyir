//
// A test is a function named `checkSomething() -> double`. It measures something and returns HOW FAR
// OFF the measurement was from what it should be: 0 means the check passed, and anything more than

}

// The height of the sine wave of frequency `hz` hidden in `count` samples that start at `start`:
// the samples are multiplied by a sine and by a cosine of that frequency and added up (a
// one-frequency Fourier transform). Exact when the stretch holds a whole number of cycles and
// nothing else is playing.
func amplitudeAt(data: ptr<float>, start: i64, count: i64, rate: i64, hz: float) -> float {
    var re: double = 0.0
    var im: double = 0.0
    for (k = 0; k < count; k = k + 1) {
        let angle: double = 6.283185307179586 * hz.toDouble() * (start + k).toDouble() / rate.toDouble()
        re = re + data[start + k].toDouble() * cos(angle)
        im = im + data[start + k].toDouble() * sin(angle)
    }
    return (2.0 * sqrt(re * re + im * im) / count.toDouble()).toFloat()
}
